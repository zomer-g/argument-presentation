API Documentation

REST API for managing cases, documents, and chapters

Base URL: http://localhost:8080/api

Cases

GET /api/cases List all cases

Response

{ "cases": [{ "id": "omar-duanias", "title": "...", "subtitle": "...", "icon": "fa-gavel", "documents": [...] }] }

curl

curl http://localhost:8080/api/cases
GET /api/cases/:caseId Get single case

Response

{ "id": "omar-duanias", "title": "...", "subtitle": "...", "icon": "fa-gavel", "documents": [...] }

curl

curl http://localhost:8080/api/cases/omar-duanias
POST /api/cases Create a new case

Request Body

{ "id": "my-case", "title": "Case Title", "subtitle": "Case Number", "icon": "fa-gavel", "documents": [] }

curl

curl -X POST http://localhost:8080/api/cases \ -H "Content-Type: application/json" \ -d '{"id":"my-case","title":"Case Title","subtitle":"123/2025","documents":[]}'
PUT /api/cases/:caseId Update case metadata

Request Body

{ "title": "New Title", "subtitle": "New Subtitle", "icon": "fa-balance-scale" }

curl

curl -X PUT http://localhost:8080/api/cases/my-case \ -H "Content-Type: application/json" \ -d '{"title":"Updated Title"}'
DELETE /api/cases/:caseId Delete case and all its files

curl

curl -X DELETE http://localhost:8080/api/cases/my-case

Documents

GET /api/cases/:caseId/docs List documents in a case

curl

curl http://localhost:8080/api/cases/omar-duanias/docs
POST /api/cases/:caseId/docs Add a document to a case

Request Body

{ "id": "my-doc", "title": "Document Title", "date": "2025" }

curl

curl -X POST http://localhost:8080/api/cases/omar-duanias/docs \ -H "Content-Type: application/json" \ -d '{"id":"my-doc","title":"Document Title","date":"2025"}'
GET /api/cases/:caseId/docs/:docId Get full document (meta, entities, chapters)

curl

curl http://localhost:8080/api/cases/omar-duanias/docs/hearing-arguments
PUT /api/cases/:caseId/docs/:docId Replace entire document JSON

Request Body

{ "meta": {...}, "entities": {...}, "chapters": [...] }
PATCH /api/cases/:caseId/docs/:docId/meta Update document metadata

Request Body

{ "title": "New Document Title", "author": "Name" }
PATCH /api/cases/:caseId/docs/:docId/entities Update document entities

Request Body

{ "people": [...], "organizations": [...] }
DELETE /api/cases/:caseId/docs/:docId Delete document and its files

curl

curl -X DELETE http://localhost:8080/api/cases/omar-duanias/docs/my-doc

Chapters

GET /api/cases/:caseId/docs/:docId/chapters List all chapters

curl

curl http://localhost:8080/api/cases/omar-duanias/docs/hearing-arguments/chapters
GET /api/cases/:caseId/docs/:docId/chapters/:chapterId Get single chapter

curl

curl http://localhost:8080/api/cases/omar-duanias/docs/hearing-arguments/chapters/intro
POST /api/cases/:caseId/docs/:docId/chapters Add a new chapter

Request Body

{ "id": "new-chapter", "title": "Chapter Title", "icon": "fa-bookmark", "content": "<p>Chapter HTML content</p>", "after": "intro" // optional: insert after this chapter }

curl

curl -X POST http://localhost:8080/api/cases/omar-duanias/docs/hearing-arguments/chapters \ -H "Content-Type: application/json" \ -d '{"id":"new-ch","title":"New Chapter","content":"<p>Content</p>"}'
PUT /api/cases/:caseId/docs/:docId/chapters/:chapterId Update chapter content/title/icon

Request Body

{ "title": "New Title", "icon": "fa-gavel", "content": "<p>Updated content</p>" }
DELETE /api/cases/:caseId/docs/:docId/chapters/:chapterId Delete a chapter

curl

curl -X DELETE http://localhost:8080/api/cases/omar-duanias/docs/hearing-arguments/chapters/new-ch
PATCH /api/cases/:caseId/docs/:docId/chapters/reorder Reorder chapters

Request Body

{ "order": ["chapter-2", "chapter-1", "chapter-3"] }

curl

curl -X PATCH http://localhost:8080/api/cases/omar-duanias/docs/hearing-arguments/chapters/reorder \ -H "Content-Type: application/json" \ -d '{"order":["personal","intro","background"]}'
